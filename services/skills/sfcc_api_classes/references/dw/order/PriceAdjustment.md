# dw.order.PriceAdjustment

## Overview
Represents an adjustment to the price of an order (product, shipping, or order-level).

## Description
PriceAdjustment models discounts and other adjustments generated by promotions or custom code. It exposes many read-only properties describing the source promotion, AB test, quantities, prorated prices and reason codes, and methods to inspect or modify certain attributes (for custom adjustments).

```ts
declare class PriceAdjustment extends dw.order.LineItem {
    /** The B2C Commerce AB-test this price adjustment is associated with. @readonly */
    ABTest: dw.campaign.ABTest

    /** The ID of the AB-test related to this price adjustment. @readonly */
    ABTestID: string

    /** The AB-test segment this price adjustment is associated with. @readonly */
    ABTestSegment: dw.campaign.ABTestSegment

    /** The ID of the AB-test segment related to this price adjustment. @readonly */
    ABTestSegmentID: string

    /** The Discount instance describing the discount applied. @readonly */
    appliedDiscount: dw.campaign.Discount

    /** True if the price adjustment was generated by an AB-test. @readonly */
    basedOnABTest: boolean

    /** True if the price adjustment was generated by a campaign. @readonly */
    basedOnCampaign: boolean

    /** True if the price adjustment was generated by a coupon. @readonly */
    basedOnCoupon: boolean

    /** The campaign this price adjustment is associated with. @readonly */
    campaign: dw.campaign.Campaign

    /** The campaign ID related to this price adjustment. @readonly */
    campaignID: string

    /** The coupon line item related to this price adjustment. @readonly */
    couponLineItem: dw.order.CouponLineItem

    /** The name of the user who created the price adjustment. @readonly */
    createdBy: string

    /** True if created by custom script. @readonly */
    custom: boolean

    /** True if this was added manually by a user. */
    manual: boolean

    /** The promotion associated with this price adjustment. @readonly */
    promotion: dw.campaign.Promotion

    /** The promotion ID related to this price adjustment. @readonly */
    promotionID: string

    /** Map of prorated prices per product line item. @readonly */
    proratedPrices: dw.util.Map

    /** Number of items this price adjustment applies to. @readonly */
    quantity: number

    /** The reason code of the price adjustment. @readonly */
    reasonCode: dw.value.EnumValue

    /** Returns the ABTest object. */
    getABTest(): dw.campaign.ABTest

    /** Returns the ABTest ID. */
    getABTestID(): string

    /** Returns the ABTestSegment object. */
    getABTestSegment(): dw.campaign.ABTestSegment

    /** Returns the ABTestSegment ID. */
    getABTestSegmentID(): string

    /** Returns the applied Discount instance. */
    getAppliedDiscount(): dw.campaign.Discount

    /** Returns the Campaign object. */
    getCampaign(): dw.campaign.Campaign

    /** Returns the Campaign ID. */
    getCampaignID(): string

    /** Returns the CouponLineItem if any. */
    getCouponLineItem(): dw.order.CouponLineItem

    /** Returns the creator name. */
    getCreatedBy(): string

    /** Returns the Promotion object. */
    getPromotion(): dw.campaign.Promotion

    /** Returns the Promotion ID. */
    getPromotionID(): string

    /** Returns prorated prices map. */
    getProratedPrices(): dw.util.Map

    /** Returns the quantity this adjustment applies to. */
    getQuantity(): number

    /** Returns the reason code EnumValue. */
    getReasonCode(): dw.value.EnumValue

    /** True if based on AB test. */
    isBasedOnABTest(): boolean

    /** Deprecated: True if based on campaign. */
    isBasedOnCampaign(): boolean

    /** True if based on coupon. */
    isBasedOnCoupon(): boolean

    /** True if custom. */
    isCustom(): boolean

    /** True if manual. */
    isManual(): boolean

    /** Marks the PriceAdjustment as manual/non-manual. */
    setManual(aFlag: boolean): void

    /** Set the reason code using internal non-localizable value. */
    setReasonCode(reasonCode: string): void
}
```
